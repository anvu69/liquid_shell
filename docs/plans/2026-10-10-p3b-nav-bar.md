- [ ] **Step 2: Native probe — re-enable the interactive pop and report progress per frame**

Create `liquid_shell_ios/ios/liquid_shell_ios/Sources/liquid_shell_ios/S2Probe.swift`:

```swift
import QuartzCore
import UIKit

/// SPIKE S2 (P3b-2 Task 1). Never commit. Reports the native interactive
/// pop's progress once per display frame, over Pigeon (`events`) and over
/// a C callback that Dart registers through FFI.
@available(iOS 26.0, *)
final class S2Probe: NSObject, UIGestureRecognizerDelegate {
  static var ffiCallback: (@convention(c) (Int32, Double, Double) -> Void)?
  weak var nav: UINavigationController?
  let tab: Int
  let send: (Int, Double, Double) -> Void   // phase, progress, timestamp
  private var link: CADisplayLink?

  init(nav: UINavigationController, tab: Int, send: @escaping (Int, Double, Double) -> Void) {
    self.nav = nav
    self.tab = tab
    self.send = send
    super.init()
    for recognizer in [nav.interactivePopGestureRecognizer, nav.interactiveContentPopGestureRecognizer] {
      recognizer?.isEnabled = true
      recognizer?.addTarget(self, action: #selector(changed(_:)))
    }
  }

  @objc private func changed(_ recognizer: UIGestureRecognizer) {
    switch recognizer.state {
    case .began:
      report(phase: 0, progress: 0)
      link = CADisplayLink(target: self, selector: #selector(frame(_:)))
      link?.preferredFrameRateRange = CAFrameRateRange(minimum: 60, maximum: 120, preferred: 120)
      link?.add(to: .main, forMode: .common)
    case .ended, .cancelled, .failed:
      guard let coordinator = nav?.transitionCoordinator else { return finish(cancelled: true) }
      coordinator.notifyWhenInteractionChanges { [weak self] context in
        self?.finish(cancelled: context.isCancelled)
      }
    default:
      break
    }
  }

  @objc private func frame(_ link: CADisplayLink) {
    guard let coordinator = nav?.transitionCoordinator else { return }
    report(phase: 1, progress: Double(coordinator.percentComplete))
  }

  private func finish(cancelled: Bool) {
    link?.invalidate()
    link = nil
    report(phase: cancelled ? 3 : 2, progress: cancelled ? 0 : 1)
  }

  private func report(phase: Int, progress: Double) {
    let now = CACurrentMediaTime()
    NSLog("[s2] native phase=%d progress=%.4f t=%.6f", phase, progress, now)
    S2Probe.ffiCallback?(Int32(phase), progress, now)
    send(phase, progress, now)
  }
}

@_cdecl("liquid_shell_s2_register")
public func liquidShellS2Register(_ callback: @convention(c) (Int32, Double, Double) -> Void) {
  if #available(iOS 26.0, *) { S2Probe.ffiCallback = callback }
}
```

```dart
// SPIKE S2 (P3b-2 Task 1). Never commit.
import 'dart:ffi';
import 'dart:developer' as developer;

import 'package:flutter/cupertino.dart';
import 'package:flutter/scheduler.dart';

/// A Cupertino page route whose transition controller the native back
/// swipe drives. Flutter's own back gesture is off: the native one owns it.
class S2Route<T> extends PageRoute<T> with CupertinoRouteTransitionMixin<T> {
  S2Route({required this.builder});

  final WidgetBuilder builder;

  static S2Route<Object?>? current;

  @override
  Widget buildContent(BuildContext context) => builder(context);

  @override
  String? get title => null;

  @override
  bool get maintainState => true;

  @override
  bool get popGestureEnabled => false;

  @override
  void install() {
    super.install();
    current = this;
  }

  void began() => navigator!.didStartUserGesture();

  void progress(double value) => controller!.value = 1 - value;

  void ended({required bool commit}) {
    if (commit) {
      navigator!.pop();
    } else {
      controller!.forward();
    }
    navigator!.didStopUserGesture();
  }
}

/// Logs one line per event: transport, phase, native progress, native
/// time, Dart's controller value after applying it, Dart's frame time.
void s2Apply(String transport, int phase, double progress, double nativeTime) {
  final route = S2Route.current;
  if (route == null) return;
  switch (phase) {
    case 0:
      route.began();
    case 1:
      route.progress(progress);
    case 2:
      route.ended(commit: true);
    case 3:
      route.ended(commit: false);
  }
  developer.log(
    '[s2] dart $transport phase=$phase native=$progress '
    'value=${route.controller?.value} nt=$nativeTime '
    'frame=${SchedulerBinding.instance.currentSystemFrameTimeStamp.inMicroseconds}',
  );
}

@Native<Void Function(Pointer<NativeFunction<Void Function(Int32, Double, Double)>>)>(
  symbol: 'liquid_shell_s2_register',
)
external void _register(
  Pointer<NativeFunction<Void Function(Int32, Double, Double)>> callback,
);

NativeCallable<Void Function(Int32, Double, Double)>? _callable;

/// FFI transport: native calls Dart synchronously on the platform thread
/// (merged threads, Flutter ≥ 3.29).
void s2UseFfi() {
  _callable ??= NativeCallable<Void Function(Int32, Double, Double)>.isolateLocal(
    (int phase, double progress, double time) =>
        s2Apply('ffi', phase, progress, time),
  );
  _register(_callable!.nativeFunction);
}

/// Pigeon transport: the spike's debug string on onSearchSubmitted.
bool s2HandleSubmitted(String text) {
  if (!text.startsWith('s2|')) return false;
  final parts = text.split('|');
  s2Apply('pigeon', int.parse(parts[1]), double.parse(parts[2]), double.parse(parts[3]));
  return true;
}
```

```dart
import 'dart:io';

/// Pairs each native frame line with the next Dart line of the same phase
/// and counts frames where Dart's applied value lags the native progress by
/// more than one or two frames' worth of progress.
void main(List<String> args) {
  final lines = File(args.single).readAsLinesSync();
  final native = RegExp(r'\[s2\] native phase=1 progress=([\d.]+) t=([\d.]+)');
  final dart = RegExp(r'\[s2\] dart \w+ phase=1 native=([\d.]+) value=([\d.]+)');
  final nativeProgress = <double>[];
  final dartValues = <double>[];
  for (final line in lines) {
    final n = native.firstMatch(line);
    if (n != null) nativeProgress.add(double.parse(n[1]!));
    final d = dart.firstMatch(line);
    if (d != null) dartValues.add(1 - double.parse(d[2]!));
  }
  final count = nativeProgress.length < dartValues.length ? nativeProgress.length : dartValues.length;
  var within1 = 0;
  var over2 = 0;
  for (var i = 0; i < count; i++) {
    // Frame-equivalent distance: how many native frames back the Dart value is.
    var lag = 0;
    while (i - lag > 0 && (nativeProgress[i - lag] - dartValues[i]).abs() > 0.004) {
      lag++;
      if (lag > 3) break;
    }
    if (lag <= 1) within1++;
    if (lag > 2) over2++;
  }
  stdout.writeln('frames=$count within1=${(100 * within1 / count).toStringAsFixed(1)}% over2=$over2');
}
```

```dart
import 'package:flutter/widgets.dart';

/// A button at the trailing end of a page's navigation bar (for example a
/// profile avatar). Natively a glass bar button item; in the Flutter bar a
/// 44pt glass circle.
@immutable
class LiquidPageAction {
  /// Creates an action.
  const LiquidPageAction({
    required this.icon,
    required this.semanticLabel,
    required this.onPressed,
    this.sfSymbol,
  });

  /// The icon of the Flutter bar's circle.
  final Widget icon;

  /// Screen-reader label and tooltip.
  final String semanticLabel;

  /// Called on tap.
  final VoidCallback onPressed;

  /// SF Symbol name for the native bar. Without it, the page draws the
  /// Flutter bar, also on iOS 26, and logs one line in debug.
  final String? sfSymbol;

  /// Field by field; [icon] and [onPressed] compare by identity.
  @override
  bool operator ==(Object other) =>
      other is LiquidPageAction &&
      other.icon == icon &&
      other.semanticLabel == semanticLabel &&
      other.onPressed == onPressed &&
      other.sfSymbol == sfSymbol;

  @override
  int get hashCode => Object.hash(icon, semanticLabel, onPressed, sfSymbol);
}

/// Whether every action can be a native bar button item. Pure.
bool pageActionsDescribable(List<LiquidPageAction> actions) =>
    actions.every((a) => a.sfSymbol != null);

/// The debug line for a page that keeps the Flutter bar because an action
/// lacks an SF Symbol, or null when none does. Pure.
String? pageActionHint(String title, List<LiquidPageAction> actions) {
  final missing = [
    for (final a in actions)
      if (a.sfSymbol == null) '"${a.semanticLabel}"',
  ];
  if (missing.isEmpty) return null;
  final noun = missing.length == 1 ? 'action' : 'actions';
  return 'liquid_shell: page "$title" draws the Flutter bar where the native '
      'bar is available: no sfSymbol on $noun ${missing.join(', ')}.';
}
```

```swift
      item.accessibilityLabel = action.title

    let current = navigationItem.rightBarButtonItems ?? []
    let same =
      current.count == items.count
      && zip(current, items.reversed()).allSatisfy {
        $0.accessibilityLabel == $1.accessibilityLabel && $0.image == $1.image

  }
```

```dart
/// A page route with the iOS push transition whose back swipe the native
/// navigation bar can drive (spec §10.6): on iOS 26 native chrome, UIKit's
/// interactive pop moves the native bar and this route's page together,
/// frame by frame. Elsewhere it is a `CupertinoPageRoute`.
class LiquidPageRoute<T> extends PageRoute<T>
    with CupertinoRouteTransitionMixin<T> {
  /// Creates the route.
  LiquidPageRoute({
    required this.builder,
    this.title,
    super.settings,
    this.maintainState = true,
    super.fullscreenDialog,
  });

  /// Builds the page.
  final WidgetBuilder builder;

  @override
  final String? title;

  @override
  final bool maintainState;

  bool _nativeDriven = false;

  @override
  Widget buildContent(BuildContext context) => builder(context);

  /// Off while the native bar owns the swipe: two back gestures would
  /// fight over the same edge.
  @override
  bool get popGestureEnabled => !_nativeDriven && super.popGestureEnabled;
}

/// The shell turns the native swipe on for the top page of a tab with a
/// native bar. Not exported.
void setNativeDriven(LiquidPageRoute<Object?> route, {required bool on}) =>
    route._nativeDriven = on;

/// UIKit's interactive pop began.
void beginNativeBackGesture(LiquidPageRoute<Object?> route) =>
    route.navigator?.didStartUserGesture();

/// UIKit's pop is [progress] (0 → 1) of the way.
void updateNativeBackGesture(LiquidPageRoute<Object?> route, double progress) {
  final controller = route.controller;
  if (controller == null || !progress.isFinite) return;
  controller.value = 1 - progress.clamp(0.0, 1.0);
}

/// UIKit's pop ended: [commit] pops the route from where the finger left
/// it (a `PopScope` may refuse: then the page returns); otherwise it
/// returns.
Future<void> endNativeBackGesture(
  LiquidPageRoute<Object?> route, {
  required bool commit,
}) async {
  final navigator = route.navigator;
  if (navigator == null) return;
  var popped = false;
  if (commit) popped = await navigator.maybePop();
  if (!popped) await route.controller?.forward();
  navigator.didStopUserGesture();
}
```

```swift
import QuartzCore
import UIKit

/// Reports UIKit's interactive pop to Dart once per display frame (spec
/// §10.6, the S2 go path). Pure enough to unit-test: the display link and
/// the transition coordinator stay in `ShellNavController`.
final class BackGestureReporter {
  private let send: (NativeGesturePhase, Double) -> Void

  init(send: @escaping (NativeGesturePhase, Double) -> Void) {
    self.send = send
  }

  func began() { send(.began, 0) }

  func frame(percent: Double) {
    guard percent.isFinite else { return }
    send(.changed, min(max(percent, 0), 1))
  }

  func finished(cancelled: Bool) { send(cancelled ? .cancelled : .ended, cancelled ? 0 : 1) }
}
```

```swift
    case .began:

        self?.reporter?.finished(cancelled: context.isCancelled)
      }
    default:
      break
    }
  }

  @objc private func gestureFrame() {
    guard let coordinator = transitionCoordinator else { return }
    reporter?.frame(percent: Double(coordinator.percentComplete))
  }
```

```dart
    return (inlineOpacity: 0, largeShift: 0);
  }

  final fadeStart = largeExtent - 16;
  final opacity = ((offset - fadeStart) / 16).clamp(0.0, 1.0);
  return (inlineOpacity: opacity, largeShift: -shift);
}
```
